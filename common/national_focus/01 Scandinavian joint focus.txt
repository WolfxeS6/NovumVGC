			is_in_faction = no
			is_subject = no
			has_capitulated = no

				limit = { NOT = { original_tag = SWE } }
				SWE = {
					exists = yes
					NVX_has_government_system_democratic = yes
					is_in_faction = no
					is_subject = no
					has_capitulated = no
				}
				NOT = {
					has_war_with = SWE
					has_wargoal_against = SWE
					is_justifying_wargoal_against = SWE
				}
			}
			if = {
				limit = { NOT = { original_tag = DEN } }
				DEN = {
					exists = yes
					NVX_has_government_system_democratic = yes
					is_in_faction = no
					is_subject = no
					has_capitulated = no
				}
				NOT = {
					has_war_with = DEN
					has_wargoal_against = DEN
					is_justifying_wargoal_against = DEN
				}
			}
			if = {
				limit = { NOT = { original_tag = NOR } }
				NOR = {
					exists = yes
					NVX_has_government_system_democratic = yes
					is_in_faction = no
					is_subject = no
					has_capitulated = no
				}
				NOT = {
					has_war_with = NOR
					has_wargoal_against = NOR
					is_justifying_wargoal_against = NOR
				}
			}
		}
		if = {
			limit = { NOT = { original_tag = SWE } }
			SWE = {
				OR = {
					has_completed_focus = SWE_welfare_and_markets
					has_completed_focus = SWE_to_free_the_people
				}
			}
		}
		if = {
			limit = { NOT = { original_tag = DEN } }
			DEN = {
				OR = { 
					has_completed_focus = DEN_welfare_state
					has_completed_focus = DEN_Danish_culture
				}
			}
			set_temp_variable = { target = DEN }
			at_peace_or_same_enemies_as_TARGET = yes
		}
		if = {
			limit = { NOT = { original_tag = NOR } }
			NOR = {
				has_completed_focus = NOR_reinforce_the_constitutional_monarchy
				focus_progress = { focus = NOR_leave_nato progress > 0 }
			}
		}
	}

	ai_will_do = {
		factor = 1000
	}

	completion_reward_joint_originator = {
		effect_tooltip = {
			create_faction_from_template = { template = faction_template_generic name = nordic_council }
			add_ideas = SCA_nordic_council_idea
		}
		hidden_effect = {
			set_country_flag = SCA_nordic_council_accepted
			save_event_target_as = SCA_nordic_council_leader
			random_other_country = {
				limit = {
					OR = {
						original_tag = SWE
						original_tag = NOR
						original_tag = DEN
					}
					is_subject = no
				}

			}
		}
	}
	completion_reward_joint_member = {
		effect_tooltip = {
			country_event = NVGC_scandinavia_formation_events.1
			custom_effect_tooltip = if_they_accept_tt
			add_ideas = SCA_nordic_council_idea
			add_to_faction = PREV
		}
	}
